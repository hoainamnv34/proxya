#!/bin/bash

# Danh sách các database cần export
DATABASES=("a" "c" "d" "e")

# Đường dẫn lưu dữ liệu
BACKUP_DIR="/backup/mongodb/$(date +%Y%m%d_%H%M%S)"
mkdir -p "$BACKUP_DIR"

# Thực hiện export từng database
for DB in "${DATABASES[@]}"; do
  echo "Exporting database: $DB"
  mongodump --db="$DB" --out="$BACKUP_DIR"
done

echo "✅ Backup completed at $BACKUP_DIR"

#!/bin/bash

# Thư mục chứa dữ liệu backup (ví dụ: /backup/mongodb/20240529_150000)
BACKUP_DIR="/backup/mongodb/20240529_150000"

# Danh sách các database cần restore
DATABASES=("a" "c" "d" "e")

# Thông tin kết nối MongoDB (bỏ nếu không cần auth)
MONGO_HOST="localhost"
MONGO_PORT="27017"
# MONGO_USER="your_user"
# MONGO_PASS="your_password"
# MONGO_AUTH_DB="admin"

# Restore từng database
for DB in "${DATABASES[@]}"; do
  DB_PATH="$BACKUP_DIR/$DB"
  if [ -d "$DB_PATH" ]; then
    echo "Restoring database: $DB"

    mongorestore \
      --host="$MONGO_HOST" \
      --port="$MONGO_PORT" \
      --db="$DB" \
      --drop \
      "$DB_PATH"

    # Nếu dùng auth, dùng lệnh sau thay thế:
    # mongorestore --host="$MONGO_HOST" --port="$MONGO_PORT" \
    #   --username="$MONGO_USER" --password="$MONGO_PASS" --authenticationDatabase="$MONGO_AUTH_DB" \
    #   --db="$DB" --drop "$DB_PATH"

  else
    echo "⚠️ Không tìm thấy thư mục backup cho database: $DB"
  fi
done

echo "✅ Restore hoàn tất."
